class TrieNode:

    self.children = {}

      self._build_trie(wordDict)
      res = []
      curr, curr_words = self.root, []
      
      dfs(idx: int, prev_idx: int, acc: List[str]):
   		if idx == len(s):

          curr = curr.children[s[idx]]
        
        idx += 1
